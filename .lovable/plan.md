```ts
useEffect(() => {
  if (!grade || !subject) { setDbTests([]); return; }
  if (grade === "5") { setDbTests([]); return; }   // ← новое
  loadPublishedTestsForGradeSubject(grade, subject)
    .then(setDbTests).catch(() => setDbTests([]));

```
